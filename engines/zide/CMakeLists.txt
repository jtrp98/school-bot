cmake_minimum_required(VERSION 3.16)
project(zide_engine LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(zide_core
  src/protocol.cpp src/transport.cpp src/transport_hid.cpp src/transport_serial.cpp
  src/discovery.cpp src/actions.cpp src/request.cpp src/line_reader.cpp
)
target_include_directories(zide_core PUBLIC include third_party PRIVATE third_party/hidapi)
if(WIN32)
  target_sources(zide_core PRIVATE third_party/hidapi/windows/hid.c)
  target_link_libraries(zide_core PUBLIC setupapi advapi32)
  target_compile_definitions(zide_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN HIDAPI_STATIC)
endif()
if(MSVC)
  target_compile_options(zide_core PRIVATE /utf-8)
endif()
add_executable(zide-engine src/main.cpp)
target_link_libraries(zide-engine PRIVATE zide_core)
if(MSVC)
  target_link_options(zide-engine PRIVATE "$<$<CONFIG:Release>:/DEBUG:NONE>")
elseif(NOT APPLE)
  target_link_options(zide-engine PRIVATE "$<$<CONFIG:Release>:-s>")
endif()
enable_testing()
add_executable(zide-protocol-tests tests/protocol_test.cpp)
target_link_libraries(zide-protocol-tests PRIVATE zide_core)
add_test(NAME zide-protocol COMMAND zide-protocol-tests)
install(TARGETS zide-engine RUNTIME DESTINATION .)
install(FILES engine.json THIRD-PARTY-NOTICES.md DESTINATION .)
install(FILES third_party/hidapi/LICENSE-bsd.txt DESTINATION . RENAME HIDAPI-LICENSE.txt)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION . RENAME JSON-LICENSE.txt)
